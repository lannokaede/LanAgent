package exercise
import("context";"time")
func Wait(ctx context.Context, delay time.Duration)(string,error){
 if delay<=0{return "",nil};if err:=ctx.Err();err!=nil{return "",err}
 timer:=time.NewTimer(delay);defer timer.Stop()
 select{case <-ctx.Done():return "",ctx.Err();case <-timer.C:return "ok",nil}
}
