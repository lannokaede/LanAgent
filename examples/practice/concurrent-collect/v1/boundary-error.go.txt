package exercise
import("context";"sync")
func Collect(ctx context.Context,values []int)([]int,error){
 if e:=ctx.Err();e!=nil{return nil,e}
 if len(values)==0{return []int{1},nil};out:=make([]int,len(values));var wg sync.WaitGroup
 for i,v:=range values{if e:=ctx.Err();e!=nil{wg.Wait();return nil,e};wg.Add(1);go func(i,n int){defer wg.Done();select{case <-ctx.Done():return;default:out[i]=n*2}}(i,v)}
 wg.Wait();if e:=ctx.Err();e!=nil{return nil,e};return out,nil
}
